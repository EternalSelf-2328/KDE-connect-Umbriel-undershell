if(PkgConfig_FOUND)
  pkg_check_modules(WAYLAND_CLIENT QUIET IMPORTED_TARGET wayland-client)
  pkg_get_variable(WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
  find_program(WAYLAND_SCANNER wayland-scanner)
  if(WAYLAND_CLIENT_FOUND AND WAYLAND_SCANNER AND EXISTS "${WAYLAND_PROTOCOLS_DIR}/staging/ext-image-copy-capture/ext-image-copy-capture-v1.xml" AND EXISTS "${WAYLAND_PROTOCOLS_DIR}/staging/ext-image-capture-source/ext-image-capture-source-v1.xml" AND EXISTS "${WAYLAND_PROTOCOLS_DIR}/staging/ext-foreign-toplevel-list/ext-foreign-toplevel-list-v1.xml")
    set(UMBRIEL_SCREEN_ENABLED TRUE)
    foreach(protocol ext-image-copy-capture ext-image-capture-source ext-foreign-toplevel-list)
      set(xml "${WAYLAND_PROTOCOLS_DIR}/staging/${protocol}/${protocol}-v1.xml")
      add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-client.h" COMMAND ${WAYLAND_SCANNER} client-header "${xml}" "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-client.h" DEPENDS "${xml}")
      add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-code.c" COMMAND ${WAYLAND_SCANNER} private-code "${xml}" "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-code.c" DEPENDS "${xml}")
      target_sources(connect-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-client.h" "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-code.c")
    endforeach()
    target_include_directories(connect-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(connect-core PUBLIC PkgConfig::WAYLAND_CLIENT)
    target_compile_definitions(connect-core PUBLIC UMBRIEL_SCREEN)
  endif()
  pkg_check_modules(GIO QUIET IMPORTED_TARGET gio-unix-2.0)
  if(GIO_FOUND)
    target_link_libraries(connect-core PUBLIC PkgConfig::GIO)
    target_compile_definitions(connect-core PUBLIC UMBRIEL_GIO)
  endif()
  pkg_check_modules(PULSE QUIET IMPORTED_TARGET libpulse)
  if(PULSE_FOUND)
    target_link_libraries(connect-core PUBLIC PkgConfig::PULSE)
    target_compile_definitions(connect-core PUBLIC UMBRIEL_PULSE)
  endif()
  pkg_check_modules(WEBP QUIET IMPORTED_TARGET libwebp)
  if(WEBP_FOUND)
    target_link_libraries(connect-core PUBLIC PkgConfig::WEBP)
    target_compile_definitions(connect-core PUBLIC UMBRIEL_WEBP)
  endif()
endif()

if(UMBRIEL_REMOTE)
  find_package(OpenSSL 3 QUIET)
  if(OpenSSL_FOUND)
    add_library(connect-remote native/remoteauth.cpp native/remoteserver.cpp native/pointer.cpp)
    target_link_libraries(connect-remote PUBLIC connect-core OpenSSL::Crypto)
    target_compile_options(connect-remote PRIVATE -Wall -Wextra -Wpedantic)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(XTEST QUIET IMPORTED_TARGET xtst x11)
      pkg_check_modules(FAKEKEY QUIET IMPORTED_TARGET libfakekey)
      if(FAKEKEY_FOUND AND XTEST_FOUND)
        target_link_libraries(connect-remote PUBLIC PkgConfig::FAKEKEY)
        target_compile_definitions(connect-remote PUBLIC UMBRIEL_FAKEKEY)
      endif()
      if(XTEST_FOUND)
        target_link_libraries(connect-remote PUBLIC PkgConfig::XTEST)
        target_compile_definitions(connect-remote PUBLIC UMBRIEL_XTEST)
      endif()
    endif()
    add_executable(umbriel-connect-remote native/remotemain.cpp)
    target_link_libraries(umbriel-connect-remote PRIVATE connect-remote connect-design)
    target_compile_options(umbriel-connect-remote PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS umbriel-connect-remote RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(PROGRAMS scripts/remote-session.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME umbriel-remote-session)
    if(BUILD_TESTING)
      find_package(Python3 COMPONENTS Interpreter QUIET)
      if(Python3_FOUND)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import cryptography" RESULT_VARIABLE UMBRIEL_CRYPTOGRAPHY)
        if(UMBRIEL_CRYPTOGRAPHY EQUAL 0)
          add_test(NAME remote-tls COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/private-session.sh ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/remote.py --binary $<TARGET_FILE:umbriel-connect-remote>)
        endif()
        add_test(NAME remote-session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/remote-session.py)
      endif()
    endif()
  else()
    message(STATUS "OpenSSL development files unavailable; remote channel disabled")
  endif()
endif()
