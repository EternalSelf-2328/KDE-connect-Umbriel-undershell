cmake_minimum_required(VERSION 3.22)
project(UmbrielConnect VERSION 0.3.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core DBus Network Widgets)
add_library(connect-core native/kdeclient.cpp native/integration.cpp native/commandstore.cpp native/wallpapers.cpp native/localremote.cpp native/filetransfer.cpp)
target_include_directories(connect-core PUBLIC native)
target_link_libraries(connect-core PUBLIC Qt6::Core Qt6::DBus Qt6::Network Qt6::Gui)
target_compile_options(connect-core PRIVATE -Wall -Wextra -Wpedantic)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(WEBP QUIET IMPORTED_TARGET libwebp)
  if(WEBP_FOUND)
    target_link_libraries(connect-core PUBLIC PkgConfig::WEBP)
    target_compile_definitions(connect-core PUBLIC UMBRIEL_WEBP)
  endif()
endif()
add_executable(umbriel-connect native/main.cpp native/window.cpp native/controlpanel.cpp native/wallpaperpanel.cpp native/sharewindow.cpp)
add_library(connect-design native/design.cpp)
target_include_directories(connect-design PUBLIC native)
target_link_libraries(connect-design PUBLIC Qt6::Widgets)
target_compile_options(connect-design PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(umbriel-connect PRIVATE connect-core connect-design)
target_compile_options(umbriel-connect PRIVATE -Wall -Wextra -Wpedantic)
configure_file(data/org.eternalself.UmbrielConnect.desktop.in org.eternalself.UmbrielConnect.desktop @ONLY)
install(TARGETS umbriel-connect RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.eternalself.UmbrielConnect.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(DIRECTORY integrations/noctalia/umbriel-connect DESTINATION ${CMAKE_INSTALL_DATADIR}/umbriel-connect/noctalia)
include(CTest)
option(UMBRIEL_REMOTE "Build experimental biometric remote channel" ON)
if(UMBRIEL_REMOTE)
  find_package(OpenSSL 3 QUIET)
  if(OpenSSL_FOUND)
    add_library(connect-remote native/remoteauth.cpp native/remoteserver.cpp native/pointer.cpp)
    target_link_libraries(connect-remote PUBLIC connect-core OpenSSL::Crypto)
    target_compile_options(connect-remote PRIVATE -Wall -Wextra -Wpedantic)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(XTEST QUIET IMPORTED_TARGET xtst x11)
      if(XTEST_FOUND)
        target_link_libraries(connect-remote PUBLIC PkgConfig::XTEST)
        target_compile_definitions(connect-remote PUBLIC UMBRIEL_XTEST)
      endif()
    endif()
    add_executable(umbriel-connect-remote native/remotemain.cpp)
    target_link_libraries(umbriel-connect-remote PRIVATE connect-remote connect-design)
    target_compile_options(umbriel-connect-remote PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS umbriel-connect-remote RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(PROGRAMS scripts/remote-session.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME umbriel-remote-session)
    if(BUILD_TESTING)
      find_package(Python3 COMPONENTS Interpreter QUIET)
      if(Python3_FOUND)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import cryptography" RESULT_VARIABLE UMBRIEL_CRYPTOGRAPHY)
        if(UMBRIEL_CRYPTOGRAPHY EQUAL 0)
          add_test(NAME remote-tls COMMAND dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/remote.py --binary $<TARGET_FILE:umbriel-connect-remote>)
        endif()
        add_test(NAME remote-session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/remote-session.py)
      endif()
    endif()
  else()
    message(STATUS "OpenSSL development files unavailable; remote channel disabled")
  endif()
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-transfers tests/transfers.cpp)
  target_link_libraries(test-transfers PRIVATE connect-core Qt6::Test)
  add_test(NAME transfers COMMAND $<TARGET_FILE:test-transfers>)
  add_executable(test-services tests/services.cpp)
  target_link_libraries(test-services PRIVATE connect-core Qt6::Test)
  add_test(NAME services COMMAND dbus-run-session -- $<TARGET_FILE:test-services>)
  if(TARGET connect-remote AND XTEST_FOUND)
    add_executable(test-pointer tests/pointer.cpp)
    target_link_libraries(test-pointer PRIVATE connect-remote Qt6::Test)
    add_test(NAME pointer COMMAND dbus-run-session -- $<TARGET_FILE:test-pointer>)
  endif()
  add_executable(test-native tests/native.cpp)
  target_link_libraries(test-native PRIVATE connect-core Qt6::Test)
  add_dependencies(test-native umbriel-connect)
  add_test(NAME native COMMAND dbus-run-session -- $<TARGET_FILE:test-native>)
  add_executable(test-design tests/design.cpp native/window.cpp native/controlpanel.cpp native/wallpaperpanel.cpp native/sharewindow.cpp)
  target_link_libraries(test-design PRIVATE connect-core connect-design Qt6::Test)
  add_test(NAME design COMMAND dbus-run-session -- ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen $<TARGET_FILE:test-design>)
endif()
